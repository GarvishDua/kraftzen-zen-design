# Every named group below repeats `Disallow: /admin`, and that repetition is
# load bearing. A crawler that matches its own named group uses ONLY that group
# and ignores `User-agent: *` entirely. An earlier version of this file listed
# the AI crawlers with a bare `Allow: /`, which silently exempted every one of
# them from the admin rule. Do not "tidy up" the duplication.

# Search crawlers
User-agent: *
Allow: /
Disallow: /admin

# AdSense. Mediapartners-Google reads page content to choose relevant ads, so
# blocking it means blank or badly targeted ads. AdsBot-Google checks landing
# page quality and, unlike every other crawler here, ignores `User-agent: *`
# outright, so it only obeys a group that names it.
User-agent: Mediapartners-Google
Allow: /
Disallow: /admin

User-agent: AdsBot-Google
Allow: /
Disallow: /admin
